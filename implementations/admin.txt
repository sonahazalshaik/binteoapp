══════════════════════════════════════════════════════════════
  ADMIN CONTROLLER → SERVICE → RESPONSE FLOW
══════════════════════════════════════════════════════════════

  Route (web.php)
       │
       ▼
  Admin\Controller@method            ← Handles request/response only
       │
       │  ┌─ Validation: FormRequest (StoreXxxRequest / UpdateXxxRequest)
       │  ├─ Gate: $this->authorize('permission')
       │  └─ Service: $this->service->method()
       │
       ▼
  App\Services\Admin\XxxService       ← All business logic here
       │
       │  ┌─ authorize(string $ability)
       │  ├─ create(Model::class, array $data)
       │  ├─ update($model, array $data)
       │  ├─ delete($model)
       │  ├─ massDelete(Model::class, array $ids)
       │  └─ handleFileUpload($model, $request, string $field)
       │
       ▼
  Response: view('admin.*') or redirect()->route()

  STANDARD ADMIN CONTROLLER TEMPLATE
══════════════════════════════════════════════════════════════

  <?php

  namespace App\Http\Controllers\Admin;

  use App\Http\Controllers\Controller;
  use App\Http\Requests\StoreXxxRequest;
  use App\Http\Requests\UpdateXxxRequest;
  use App\Models\Xxx;
  use App\Services\Admin\XxxService;

  class XxxController extends Controller
  {
      protected XxxService $service;

      public function __construct(XxxService $service)
      {
          $this->service = $service;
      }

      public function index(Request $request)
      {
          $this->service->authorize('xxx_access');

          if ($request->ajax()) {
              $query = Xxx::query()->select(sprintf('%s.*', (new Xxx)->getTable()));
              $table = \Yajra\DataTables\Facades\DataTables::of($query);
              $table->addColumn('actions', '&nbsp;');
              $table->editColumn('actions', function ($row) {
                  return view('partials.datatables_actions', [
                      'viewGate'      => 'xxx_show',
                      'editGate'      => 'xxx_edit',
                      'deleteGate'    => 'xxx_delete',
                      'route'         => 'admin.xxx',
                      'id'            => $row->id
                  ]);
              });
              $table->rawColumns(['actions']);
              return $table->make(true);
          }

          return view('admin.xxx.index');
      }

      public function create()
      {
          $this->service->authorize('xxx_create');
          return view('admin.xxx.create');
      }

      public function store(StoreXxxRequest $request)
      {
          $xxx = $this->service->create(Xxx::class, $request->validated());

          if ($request->hasFile('image')) {
              $this->service->handleFileUpload($xxx, $request, 'image');
          }

          return redirect()->route('admin.xxx.index')
              ->with('success', 'Created successfully');
      }

      public function show(Xxx $xxx)
      {
          $this->service->authorize('xxx_show');
          return view('admin.xxx.show', compact('xxx'));
      }

      public function edit(Xxx $xxx)
      {
          $this->service->authorize('xxx_edit');
          return view('admin.xxx.edit', compact('xxx'));
      }

      public function update(UpdateXxxRequest $request, Xxx $xxx)
      {
          $this->service->update($xxx, $request->validated());

          if ($request->hasFile('image')) {
              $this->service->handleFileUpload($xxx, $request, 'image');
          }

          return redirect()->route('admin.xxx.index')
              ->with('success', 'Updated successfully');
      }

      public function destroy(Xxx $xxx)
      {
          $this->service->authorize('xxx_delete');
          $this->service->delete($xxx);
          return back()->with('success', 'Deleted successfully');
      }

      public function massDestroy(Request $request)
      {
          $this->service->massDelete(Xxx::class, $request->ids);
          return response()->json(['success' => true]);
      }
  }

══════════════════════════════════════════════════════════════
  ADMIN SERVICE TEMPLATE
══════════════════════════════════════════════════════════════

  <?php

  namespace App\Services\Admin;

  use Illuminate\Support\Facades\Gate;
  use Symfony\Component\HttpFoundation\Response;

  class XxxService
  {
      public function authorize(string $ability): void
      {
          abort_if(Gate::denies($ability), Response::HTTP_FORBIDDEN, '403 Forbidden');
      }

      public function create(string $modelClass, array $data)
      {
          return $modelClass::create($data);
      }

      public function update($model, array $data)
      {
          $model->update($data);
          return $model;
      }

      public function delete($model): void
      {
          $model->delete();
      }

      public function massDelete(string $modelClass, array $ids): void
      {
          $modelClass::whereIn('id', $ids)->delete();
      }

      public function handleFileUpload($model, $request, string $field): void
      {
          $path = fileUploader($request->file($field), getFilePath($field));
          $model->update([$field => $path]);
      }
  }

  <?php

  namespace App\Http\Requests;

  use Illuminate\Foundation\Http\FormRequest;
  use Illuminate\Support\Facades\Gate;

  class StoreXxxRequest extends FormRequest
  {
      public function authorize(): bool
      {
          return Gate::allows('xxx_create');
      }

      public function rules(): array
      {
          return [
              'name'  => ['required', 'string', 'max:255'],
              'email' => ['required', 'email', 'unique:users,email'],
              'image' => ['nullable', 'image', 'mimes:jpg,jpeg,png', 'max:5120'],
          ];
      }

      public function messages(): array
      {
          return [
              'name.required' => 'The name field is required.',
          ];
      }
  }

══════════════════════════════════════════════════════════════
  ⚠  DO'S & DON'TS
══════════════════════════════════════════════════════════════

  ✅ DO's:
  • Keep existing Admin controllers exactly as they are —
    DO NOT move, rename, or refactor them unless explicitly told.
  • Apply this pattern ONLY to NEW controllers going forward.
  • Keep business logic in Services; controllers only handle
    request/response.
  • Use FormRequest classes for validation in new controllers.
  • Use Gate::authorize() or $this->authorize() for permissions.
  • Always return consistent responses: view() or redirect().

  ❌ DON'Ts:
  • DON'T touch existing controllers like AdminController,
    VideoController, UserController, DashboardController, etc.
    They work fine.
  • DON'T delete or modify existing routes in web.php.
  • DON'T change existing model relationships or table structures.
  • DON'T remove existing helper functions (gs(), activeTemplate(),
    fileUploader(), getPaginate(), getTrx(), etc.) — they are
    used everywhere.
  • DON'T replace existing helper-based logic with service-based
    logic in old controllers — only new controllers use services.
  • DON'T change the way $notify[] or ->withNotify() works in
    existing controllers — that's the project-wide standard.
  • DON'T rename existing namespaces or class names.
