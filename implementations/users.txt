══════════════════════════════════════════════════════════════
  USER CONTROLLER → SERVICE → RESPONSE FLOW
══════════════════════════════════════════════════════════════

  Route (web.php — auth middleware)
       │
       ▼
  User\Controller@method                  ← Handles request/response
       │
       │  ┌─ Auth: auth()->user() (always authenticated)
       │  ├─ Validation: $request->validate()
       │  ├─ Service: $this->userService->method(auth()->user(), $data)
       │  └─ Notifications/AdminNotification as needed
       │
       ▼
  App\Services\UserService                ← Business logic
       │
       │  ┌─ Operates on authenticated user
       │  ├─ Updates user profiles, settings, balances
       │  ├─ Manages transactions, KYC, withdrawals
       │  └─ Returns arrays/collections or bool
       │
       ▼
  Response: view(activeTemplate() . 'client.*', $data)
            or redirect()->route() with ->withNotify($notify)
            or response()->json()

  STANDARD USER CONTROLLER TEMPLATE
══════════════════════════════════════════════════════════════

  <?php

  namespace App\Http\Controllers\User;

  use App\Constants\Status;
  use App\Http\Controllers\Controller;
  use App\Services\UserService;
  use App\Services\TransactionService;
  use Illuminate\Http\Request;

  class XxxController extends Controller
  {
      protected UserService $userService;
      protected TransactionService $transactionService;

      public function __construct(
          UserService $userService,
          TransactionService $transactionService
      ) {
          $this->userService = $userService;
          $this->transactionService = $transactionService;
      }

      public function index()
      {
          $pageTitle = 'Dashboard';
          $user = auth()->user();
          $data = $this->userService->getDashboardData($user);

          return view(activeTemplate() . 'client.dashboard', array_merge(
              compact('pageTitle', 'user'),
              $data
          ));
      }

      public function store(Request $request)
      {
          $request->validate([
              'name' => 'required|string|max:255',
              'value' => 'required|numeric',
          ]);

          $result = $this->userService->updateProfile(auth()->user(), $request->all());

          if (!$result['success']) {
              $notify[] = ['error', $result['message']];
              return back()->withNotify($notify);
          }

          $notify[] = ['success', $result['message']];
          return back()->withNotify($notify);
      }

      public function transactions()
      {
          $pageTitle = 'Transaction History';
          $user = auth()->user();
          $transactions = $this->transactionService->getUserTransactions($user);

          return view(activeTemplate() . 'client.transactions', compact(
              'pageTitle', 'transactions'
          ));
      }

      public function notifications()
      {
          $pageTitle = 'Notifications';
          $user = auth()->user();
          $notifications = $this->userService->getNotifications($user);

          return view(activeTemplate() . 'client.notifications', compact(
              'pageTitle', 'notifications'
          ));
      }
  }

  USER SERVICE TEMPLATE
══════════════════════════════════════════════════════════════

  <?php

  namespace App\Services;

  use App\Models\User;
  use App\Models\Transaction;
  use App\Models\AdminNotification;
  use Illuminate\Support\Facades\Hash;

  class UserService
  {
      public function getDashboardData(User $user): array
      {
          return [
              'total_views' => $user->videos()->sum('views_count'),
              'total_earnings' => $user->transactions()
                  ->where('trx_type', '+')
                  ->sum('amount'),
              'recent_activity' => $user->transactions()
                  ->latest()->take(10)->get(),
          ];
      }

      public function updateProfile(User $user, array $data): array
      {
          if (isset($data['password'])) {
              $data['password'] = Hash::make($data['password']);
          }

          $user->update($data);

          AdminNotification::create([
              'user_id' => $user->id,
              'title' => 'Profile updated by ' . $user->username,
              'click_url' => urlPath('admin.users.detail', $user->id),
          ]);

          return ['success' => true, 'message' => 'Profile updated successfully'];
      }

      public function getNotifications(User $user)
      {
          return $user->userNotifications()->latest()->paginate(getPaginate());
      }

      public function updateFcmToken(User $user, string $token): void
      {
          \App\Models\FirebaseToken::updateOrCreate(
              ['user_id' => $user->id],
              ['token' => $token]
          );
      }
  }

  USER-SPECIFIC PATTERNS
══════════════════════════════════════════════════════════════

  2. Always check auth()->check() before accessing user
  3. User services receive the User model as first parameter
  4. AdminNotification should be created for important changes

  7. Notify pattern: $notify[] = ['type', 'message']; return back()->withNotify($notify);
