══════════════════════════════════════════════════════════════
  WEB CONTROLLER → SERVICE → RESPONSE FLOW
══════════════════════════════════════════════════════════════

  Browser Request
       │
       ▼
  Root\Controller@method                ← Handles request/response only
       │
       │  ┌─ Validation: $request->validate() or Validator::make()
       │  ├─ Auth: auth()->user() / auth()->check()
       │  ├─ Service: $this->xxxService->method()
       │  └─ Common Data: gs(), activeTemplate()
       │
       ▼
  App\Services\XxxService               ← All business logic here
       │
       │  ┌─ Queries Models (Eloquent)
       │  ├─ Encapsulates all business rules
       │  ├─ Returns arrays or collections to controller
       │  └─ Never touches request/session directly
       │
       ▼
  Response: view(activeTemplate() . 'path', $data)
            or redirect()->route() with ->withNotify($notify)
            or response()->json() for AJAX

══════════════════════════════════════════════════════════════

  <?php

  namespace App\Http\Controllers;

  use App\Http\Controllers\Controller;
  use App\Services\XxxService;
  use Illuminate\Http\Request;

  class XxxController extends Controller
  {
      protected XxxService $xxxService;

      public function __construct(XxxService $xxxService)
      {
          $this->xxxService = $xxxService;
      }

      public function index(Request $request)
      {
          $data = $this->xxxService->getAll($request);

          if ($request->ajax()) {
              return response()->json($data);
          }

          return view(activeTemplate() . 'xxx.index', $data);
      }

      public function show($id)
      {
          $data = $this->xxxService->getById($id);

          if (!$data) {
              $notify[] = ['error', 'Not found'];
              return redirect()->route('home')->withNotify($notify);
          }

          return view(activeTemplate() . 'xxx.show', $data);
      }

      public function store(Request $request)
      {
          $request->validate([
              'name' => 'required|string|max:255',
          ]);

          $result = $this->xxxService->create($request->all());

          $notify[] = ['success', 'Created successfully'];
          return redirect()->route('xxx.index')->withNotify($notify);
      }

      public function destroy($id)
      {
          $this->xxxService->delete($id);

          if (request()->ajax()) {
              return response()->json(['status' => 'success']);
          }

          $notify[] = ['success', 'Deleted successfully'];
          return back()->withNotify($notify);
      }
  }

  STANDARD WEB SERVICE TEMPLATE
══════════════════════════════════════════════════════════════

  <?php

  namespace App\Services;

  use App\Models\Xxx;
  use Illuminate\Http\Request;

  class XxxService
  {
      public function getAll(Request $request): array
      {
          $query = Xxx::query();

          if ($request->has('search')) {
              $query->where('name', 'LIKE', "%{$request->search}%");
          }

          $items = $query->latest()->paginate(getPaginate());

          return [
              'items' => $items,
              'pageTitle' => 'Xxx List',
          ];
      }

      public function getById($id): ?array
      {
          $item = Xxx::with(['user'])->find($id);
          if (!$item) {
              return null;
          }

          return [
              'item' => $item,
              'pageTitle' => $item->name,
          ];
      }

      public function create(array $data)
      {
          return Xxx::create($data);
      }

      public function delete($id): void
      {
          $item = Xxx::findOrFail($id);
          $item->delete();
      }
  }

  ⚠  DO'S & DON'TS
══════════════════════════════════════════════════════════════

  ✅ DO's:
  • Keep ALL existing root-level controllers exactly as they are.
    DO NOT move them — routes depend on current locations.
  • Apply the Web/ pattern ONLY for brand-new controllers.
  • Use activeTemplate() for all frontend view paths.
  • Use ->withNotify($notify) for flash messages.
  • Use $request->validate() for inline validation.
  • Use gs() helper for GeneralSetting access.
  • Use getPaginate() for consistent pagination.

  ❌ DON'Ts:
  • DON'T move root controllers into Web/ folder — breaks routes.
  • DON'T change existing routes in web.php.
  • DON'T rename existing controller namespaces.
  • DON'T modify the activeTemplate() helper or view paths.
  • DON'T change the $notify pattern — used project-wide.
  • DON'T remove existing helper functions.
  • DON'T touch existing model scopes (forUser(), published()).
  • DON'T change JSON response formats used by Flutter app.
  • DON'T modify BunnyWebhookController, TusProxyController,
    or CronController — they are infrastructure-only.
