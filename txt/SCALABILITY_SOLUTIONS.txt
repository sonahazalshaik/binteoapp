  ----------------------------------

  Add new cron job:

  - Users can watch videos while others upload

FIX #6: MOVE VIEW COUNTING + GeoIP TO QUEUE
================================================================================
  Priority: MEDIUM - Removes external API latency from video views
  File: app/Services/Frontend/VideoService.php

  STEP 1 - Create a new job file:
  ---------------------------------
  Create: app/Jobs/RecordVideoView.php

  <?php

  namespace App\Jobs;

  use App\Models\Video;
  use App\Models\ViewLog;
  use Illuminate\Bus\Queueable;
  use Illuminate\Contracts\Queue\ShouldQueue;
  use Illuminate\Foundation\Bus\Dispatchable;
  use Illuminate\Queue\InteractsWithQueue;
  use Illuminate\Queue\SerializesModels;

  class RecordVideoView implements ShouldQueue
  {
      use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

      public $timeout = 30;
      public $tries = 3;

      protected $videoId;
      protected $userId;
      protected $device;

      public function __construct($videoId, $userId, $device)
      {
          $this->videoId = $videoId;
          $this->userId = $userId;
          $this->device = $device;
      }

      public function handle()
      {
          $v = Video::find($this->videoId);
          if (!$v) return;

          $v->increment('views_count');
          if ($v->impressions < $v->views_count) {
              $v->update(['impressions' => $v->views_count]);
          } else {
              $v->increment('impressions');
          }

          if ($this->userId) {
              $info = getIpInfo();
              $raw = $info['country'] ?? [];
              $country = is_array($raw) ? null : (string)$raw;
              ViewLog::updateOrCreate(
                  ['user_id' => $this->userId, 'video_id' => $this->videoId],
                  ['updated_at' => now(), 'device' => $this->device, 'country' => $country]
              );
          }
      }
  }

  STEP 2 - Change VideoService.php (lines 183-201):
  --------------------------------------------------
  BEFORE:
    app()->terminating(function () use ($videoId, $userId, $device) {
        $v = Video::find($videoId);
        if (!$v) return;
        $v->increment('views_count');
        if ($v->impressions < $v->views_count) {
            $v->update(['impressions' => $v->views_count]);
        } else {
            $v->increment('impressions');
        }
        if ($userId) {
            $info = getIpInfo();
            $raw = $info['country'] ?? [];
            $country = is_array($raw) ? null : (string)$raw;
            ViewLog::updateOrCreate(
                ['user_id' => $userId, 'video_id' => $videoId],
                ['updated_at' => now(), 'device' => $device, 'country' => $country]
            );
        }
    });

  AFTER:
    RecordVideoView::dispatch($videoId, $userId, $device);

  WHAT THIS FIXES:
  - Video page loads instantly (no blocking GeoIP call)
  - View counting happens in background
  - 1000 concurrent viewers don't slow down the server
